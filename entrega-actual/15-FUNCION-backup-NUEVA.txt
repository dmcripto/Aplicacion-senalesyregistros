// Copia de seguridad semanal de las tablas importantes (ver _shared/backup.ts).
//
//   POST /functions/v1/backup   con el encabezado x-cron-secret (= EXCHANGE_CRON_SECRET)
//
// La dispara sola la función auto-close una vez por semana (y la primera vez que se despliega). Se puede forzar a mano llamándola con el mismo
// encabezado. Guarda los archivos en el depósito privado «backups» (Storage); se bajan desde Supabase > Storage > backups.

import { createClient } from "npm:@supabase/supabase-js@2";

// ─── VELTRIX · Copia de seguridad semanal ───────────────────────────────────
// Guarda, en un depósito privado de Supabase Storage («backups»), una copia comprimida de las tablas importantes: una carpeta por día
// y un archivo por tabla. Quedan las últimas 8 copias. NO se copian las claves de los exchanges ni los códigos de vinculación
// (exchange_secrets, trade_keys, telegram_link_codes, whatsapp_codes): esas se vuelven a cargar a mano si hiciera falta restaurar.

const BUCKET = "backups";
const KEEP_COPIES = 8;
const PAGE = 1000;

interface BackupTable {
  table: string;
  /** Columnas para ordenar al paginar (la clave primaria), así ninguna fila se repite ni se pierde entre páginas. */
  order: string[];
}

const BACKUP_TABLES: BackupTable[] = [
  { table: "profiles", order: ["id"] },
  { table: "trades", order: ["id"] },
  { table: "bot_settings", order: ["user_id"] },
  { table: "bot_live", order: ["user_id"] },
  { table: "live_orders", order: ["id"] },
  { table: "bot_signals", order: ["id"] },
  { table: "unlock_history", order: ["symbol", "at", "source"] },
  { table: "token_unlocks", order: ["symbol", "at"] },
  { table: "price_alerts", order: ["id"] },
  { table: "telegram_links", order: ["user_id"] },
  { table: "telegram_communities", order: ["id"] },
  { table: "exchange_connections", order: ["id"] },
  { table: "referrals", order: ["invitee_id"] },
  { table: "ref_events", order: ["id"] },
  { table: "signal_votes", order: ["trade_id", "voter"] },
];

const isoDay = (ms: number) => new Date(ms).toISOString().slice(0, 10);

/** Todas las filas de una tabla, de a páginas. */
async function fetchAllRows(supabase: any, t: BackupTable): Promise<Array<Record<string, unknown>> | null> {
  const out: Array<Record<string, unknown>> = [];
  for (let from = 0; ; from += PAGE) {
    let q = supabase.from(t.table).select("*");
    for (const col of t.order) q = q.order(col, { ascending: true });
    const { data, error } = await q.range(from, from + PAGE - 1);
    if (error) return null; // tabla que todavía no existe en esta base: se salta
    const rows = (data ?? []) as Array<Record<string, unknown>>;
    out.push(...rows);
    if (rows.length < PAGE) return out;
  }
}

async function gzip(text: string): Promise<Uint8Array> {
  const stream = new Blob([text]).stream().pipeThrough(new CompressionStream("gzip"));
  return new Uint8Array(await new Response(stream).arrayBuffer());
}

interface BackupResult {
  day: string;
  saved: Array<{ table: string; rows: number; bytes: number }>;
  skipped: string[];
  pruned: string[];
}

async function runBackup(supabase: any, now = Date.now(), tables: BackupTable[] = BACKUP_TABLES): Promise<BackupResult> {
  const day = isoDay(now);
  const res: BackupResult = { day, saved: [], skipped: [], pruned: [] };
  const store = supabase.storage.from(BUCKET);
  for (const t of tables) {
    const rows = await fetchAllRows(supabase, t);
    if (!rows) {
      res.skipped.push(t.table);
      continue;
    }
    const bytes = await gzip(JSON.stringify(rows));
    const { error } = await store.upload(`${day}/${t.table}.json.gz`, bytes, { contentType: "application/gzip", upsert: true });
    if (error) throw new Error(`No se pudo guardar ${t.table}: ${error.message}`);
    res.saved.push({ table: t.table, rows: rows.length, bytes: bytes.length });
  }
  // Se conservan las últimas KEEP_COPIES carpetas (los nombres son fechas, así que ordenan solos).
  const { data: top } = await store.list("", { limit: 100 });
  const days = ((top ?? []) as Array<{ name: string }>).map((x) => x.name).filter((n) => /^\d{4}-\d{2}-\d{2}$/.test(n)).sort();
  for (const old of days.slice(0, Math.max(0, days.length - KEEP_COPIES))) {
    const { data: files } = await store.list(old, { limit: 100 });
    const paths = ((files ?? []) as Array<{ name: string }>).map((f) => `${old}/${f.name}`);
    if (paths.length) await store.remove(paths);
    res.pruned.push(old);
  }
  return res;
}


// ─── VELTRIX · Vigilante de funciones (solo el latido) ───
// ─── VELTRIX · Vigilante de funciones ───────────────────────────────────────
// Las tareas automáticas (el bot cada 5 minutos, la sincronización de exchanges cada 10) dejan un «latido» al terminar bien.
// Si una se detiene (error, límite de CPU, tarea apagada), el vigilante avisa por Telegram y por la app a quienes administran
// el proyecto (cuentas con el bot habilitado) y avisa de nuevo cuando vuelve a funcionar.

interface WatchedJob {
  name: string;
  /** Minutos sin latido a partir de los cuales se avisa. */
  maxMin: number;
  es: string;
  en: string;
}

const WATCHED_JOBS: WatchedJob[] = [
  { name: "bot", maxMin: 20, es: "el bot automático", en: "the automatic bot" },
  { name: "exchanges", maxMin: 35, es: "la sincronización con los exchanges", en: "the exchange sync" },
];

const REMIND_MS = 3 * 3_600_000; // mientras siga caída, se repite el aviso cada 3 horas

/** Anota que la tarea terminó bien. Nunca rompe a quien la llama (si la tabla todavía no existe, no hace nada). */
async function beat(supabase: any, name: string, now = Date.now()): Promise<void> {
  try {
    await supabase.from("system_heartbeat").upsert({ name, at: new Date(now).toISOString() }, { onConflict: "name" });
  } catch {
    /* sin la tabla (migración 0067): se sigue sin vigilante */
  }
}

interface WatchdogAlert {
  job: WatchedJob;
  kind: "down" | "recovered";
  minutes: number;
}

/** Revisa los latidos; devuelve los avisos que corresponde mandar (y deja anotado cuándo se mandaron para no repetirlos). */
async function runWatchdog(supabase: any, notify: (alert: WatchdogAlert) => Promise<void>, now = Date.now(), jobs: WatchedJob[] = WATCHED_JOBS): Promise<WatchdogAlert[]> {
  const sent: WatchdogAlert[] = [];
  let rows: Array<{ name: string; at: string; alerted_at: string | null }> = [];
  try {
    const { data, error } = await supabase.from("system_heartbeat").select("name, at, alerted_at");
    if (error) return sent;
    rows = (data ?? []) as typeof rows;
  } catch {
    return sent;
  }
  for (const job of jobs) {
    const row = rows.find((r) => r.name === job.name);
    if (!row) continue; // nunca latió: puede ser una tarea que todavía no se configuró; no se avisa
    const age = now - new Date(row.at).getTime();
    const minutes = Math.round(age / 60_000);
    const stale = age > job.maxMin * 60_000;
    const alertedAt = row.alerted_at ? new Date(row.alerted_at).getTime() : null;
    if (stale && (alertedAt == null || now - alertedAt >= REMIND_MS)) {
      const alert: WatchdogAlert = { job, kind: "down", minutes };
      await notify(alert).catch(() => undefined);
      await supabase.from("system_heartbeat").update({ alerted_at: new Date(now).toISOString() }).eq("name", job.name);
      sent.push(alert);
    } else if (!stale && alertedAt != null) {
      const alert: WatchdogAlert = { job, kind: "recovered", minutes };
      await notify(alert).catch(() => undefined);
      await supabase.from("system_heartbeat").update({ alerted_at: null }).eq("name", job.name);
      sent.push(alert);
    }
  }
  return sent;
}

const watchdogText = (a: WatchdogAlert, lang: "es" | "en"): { title: string; body: string } => {
  const name = lang === "en" ? a.job.en : a.job.es;
  if (a.kind === "recovered") {
    return lang === "en" ? { title: "✅ Back to normal", body: `${name[0].toUpperCase()}${name.slice(1)} is running again.` } : { title: "✅ Todo volvió a la normalidad", body: `${name[0].toUpperCase()}${name.slice(1)} volvió a funcionar.` };
  }
  return lang === "en"
    ? { title: "⚠️ A VELTRIX task stopped", body: `${name[0].toUpperCase()}${name.slice(1)} has not run for ${a.minutes} minutes. Check the function Logs in Supabase.` }
    : { title: "⚠️ Se detuvo una tarea de VELTRIX", body: `${name[0].toUpperCase()}${name.slice(1)} no corre hace ${a.minutes} minutos. Revisá los Logs de la función en Supabase.` };
};


const admin = createClient(Deno.env.get("SUPABASE_URL")!, Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!);
const json = (body: unknown, status = 200) => new Response(JSON.stringify(body), { status, headers: { "Content-Type": "application/json" } });

function same(a: string, b: string): boolean {
  if (a.length !== b.length) return false;
  let d = 0;
  for (let i = 0; i < a.length; i++) d |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return d === 0;
}

Deno.serve(async (req) => {
  if (req.method !== "POST") return json({ ok: false, error: "Method not allowed" }, 405);
  const secret = Deno.env.get("EXCHANGE_CRON_SECRET");
  const given = req.headers.get("x-cron-secret");
  if (!secret || secret.length < 16 || !given || !same(given, secret)) return json({ ok: false, error: "No autorizado" }, 401);
  try {
    const r = await runBackup(admin);
    await beat(admin, "backup");
    return json({ ok: true, ...r });
  } catch (e) {
    console.error("backup:", e instanceof Error ? e.message : e);
    return json({ ok: false, error: e instanceof Error ? e.message : "backup" }, 500);
  }
});
